Theorem~4 of \sscite{fagerstrom2005temporal} inverts the scale-Cauchy problem into an
evolution \emph{in time} over the memory line: $\partial_t u = D^{1/\alpha}_{x,-} u$ with
boundary data $u(t, 0) = f(t)$ and $u(0, x) = 0$ --- the signaling problem. The operator there
is translation-invariant in $x$, which is available only in the homogeneous (semigroup) case.
This chapter constructs the general inversion.

The structural observation that replaces homogeneity is this: writing $H := e^{-F}$, the
Laplace-domain profiles of the field,
\[
  \hat u(s, x) = \hat f(s)\, H(sx),
\]
are \emph{dilates of a single profile} $H$, with the eigenvalue $s$ appearing as the dilation
parameter. Operators diagonal on dilation families are the subject of multiplicative harmonic
analysis, so the inversion operator lives in the \textbf{Mellin calculus} --- the
multiplicative-group counterpart of the Wendel step of Chapter~4 on the additive group.

Throughout we work in the canonical gauge, so that $\mu_{0,x}$ is the law of $x\,T_1$ with
$\Ex[e^{-sT_1}] = H(s)$; we write $\tilde g(z) := \int_0^\infty x^{z-1} g(x)\,dx$ for the
Mellin transform, $\theta := x\,\partial_x$ for the Euler operator, and
$(I^z f)(t) := \tfrac{1}{\Gamma(z)}\int_0^t (t-r)^{z-1} f(r)\,dr$ for the Riemann--Liouville
integral of complex order $z$, $\operatorname{Re} z > 0$ (\sscite{samko1993fractional}).

\begin{definition}[the standing hypothesis (H)]\label{def:standing-hypothesis}
\uses{lem:selfdecomposable-exponents,thm:main-characterization}
An admissible exponent $F$ satisfies the \emph{standing hypothesis} (H) if
$F(\infty) = \infty$ --- equivalently $b_0 > 0$ or $\int_0^\infty k(t)\,t^{-1}dt = \infty$,
i.e.\ no atom of the kernels at zero delay --- and
\[
  z_* \;:=\; \sup\bigl\{\, \zeta > 0 :\ \Ex\bigl[T_1^{-\zeta}\bigr] < \infty \,\bigr\} \;>\; 1 .
\]
\statusT
\end{definition}

The second condition is equivalent to $\int_0^\infty e^{-F(s)}\,ds < \infty$. It holds with
$z_* = \infty$ whenever $b_0 > 0$, for the stable family, and for the Bessel-K family of
\ref{prop:bessel-family}; for the Gamma family $z_* = \gamma$, so (H) requires $\gamma > 1$
there. The construction extends below $z_* = 1$ by meromorphic continuation, which we do not
pursue.

\begin{lemma}[Mellin data of the profile]\label{lem:mellin-data}
% The edge to prop:moment-criterion went with the narrowing: that node (ledger A7) is the
% criterion for POSITIVE integer moments of T_x, and the statement left here is about
% NEGATIVE moments, which z_* governs and def:standing-hypothesis supplies. The Lean proof

% through A7 is used.
\uses{def:standing-hypothesis,thm:main-characterization}
\lean{Hemigroup.SelfDecomposableExponent.mellin_profile}\leanok
Under (H), for $0 < \operatorname{Re} z < z_*$,
\[
  \tilde H(z) \;=\; \Gamma(z)\;\Ex\bigl[T_1^{-z}\bigr],
\]
with $|\tilde H(c + i\tau)| \le \Ex[T_1^{-c}]\,|\Gamma(c+i\tau)|$ for $0 < c < z_*$.
\statusT\quad\emph{(\textbf{Machine-checked}, as
\texttt{Hemigroup.SelfDecomposableExponent.mellin\_profile} and
\texttt{norm\_mellin\_profile\_le}; \texttt{\#print axioms} gives \ledger{A17} and nothing else,
inherited through \texttt{kernel} because $T_1$ is $\mu_{0,1}$. In particular \emph{no} chapter-11
interface enters --- \ledger{A12} grounds \ref{def:inversion-operator}, not this.}

\emph{\textbf{The exchange of integrals is licensed by the strip condition and by nothing else.}
Both clauses come out of one $[0,\infty]$-valued computation,
$\int\!\!\int s^{c-1}e^{-ts}\,ds\,d\mu(t) = \Gamma(c)\,\Ex[T_1^{-c}]$, which is simultaneously the
Fubini side condition, the bound, and the finiteness underneath \ref{lem:mellin-vertical}. Its
left-hand side is the total mass of $|{\cdot}|$ for the product measure, so joint integrability
holds \emph{iff} the right-hand side is finite, which is $c < z_*$. That is the shape of
\ref{prop:admissibility-criterion} again: the condition that makes the analysis go through is
characteristic rather than merely sufficient, and seeing that is what keeps a hypothesis from
being invented.}

\emph{\textbf{The first clause of (H) is load-bearing.} $F(\infty) = \infty$ is used exactly once,
to give $T_1 > 0$ almost surely --- the mass at the origin is below $\Ex[e^{-sT_1}] = e^{-F(s)}$ for
every $s$. Without it the computation above is \emph{false}, not merely unprovable: at an atom
$t = 0$ the inner integral $\int_0^\infty s^{c-1}ds$ diverges, while $\Ex[T_1^{-c}]$, an integral
over $(0,\infty)$, does not see the atom at all.}

\emph{Three design decisions were taken in writing the statement down. $T_1$ is the kernel
$\mu_{0,1}$ the construction already produces, so no new object enters and the density $\phi_1$
is \emph{not} needed --- which matters, because $\phi_1$ would drag in \ledger{A9} for a node
that does not otherwise need it. The profile is the Laplace transform rather than the density, so
Mathlib's \texttt{mellin} applies to it directly. And the negative moments are carried in
$[0,\infty]$: divergence is the whole point of $z_*$, and Chapter~9 is the cautionary tale about
departing from that convention.)}
\end{lemma}

\begin{proof}
\leanok
By Tonelli,
\[
  \tilde H(z) = \int_0^\infty s^{z-1}\,\Ex[e^{-sT_1}]\,ds
  = \Ex\Bigl[\int_0^\infty s^{z-1}e^{-sT_1}ds\Bigr]
  = \Ex[T_1^{-z}]\,\Gamma(z),
\]
the inner integral being the Gamma integral at rate $T_1$. The exchange is legitimate because the
same computation run on absolute values gives $\Gamma(c)\,\Ex[T_1^{-c}]$ with
$c = \operatorname{Re} z$, finite exactly when $c < z_*$. Since
$|T_1^{-z}| = T_1^{-\operatorname{Re} z}$, the identity also gives the stated bound.
\end{proof}

\begin{definition}[the inversion operator]\label{def:inversion-operator}
\uses{lem:mellin-data,lem:inversion-symbol,def:cascade-family}
\lean{Hemigroup.SelfDecomposableExponent.inversionOperator}\leanok
Fix $c \in (0, z_* - 1)$. For $g$ such that $\tilde g$ is defined on the line
$\operatorname{Re} z = c$ and $B(-z)\tilde g(z)$ is absolutely integrable there, set
\[
  \bigl(A g\bigr)(x) \;:=\; \frac{1}{x}\cdot \frac{1}{2\pi i}\int_{(c)} x^{-z}\, B(-z)\,\tilde g(z)\,dz
  \;=\; \Bigl(\frac{1}{x}\,B(\theta)\,g\Bigr)(x),
\]
the second expression being the functional-calculus reading, the Euler operator having Mellin
symbol $-z$. The second equality is an assertion and not a rewriting, and it is worth being
explicit about what it asserts: $B(\theta)$ is a functional calculus for a symbol with poles, so
what the display says is that there \emph{exists} a function $h$ on $(0,\infty)$ whose Mellin
transform is $B(-z)\tilde g(z)$ on the line, and that the contour integral computes $h(x)/x$.
Absolute integrability on the line is exactly the hypothesis under which the inversion integral
recovers such an $h$. Given $h$, the rest is elementary and is
\ref{lem:inversion-operator-action}; in particular the transform-level form is
$\widetilde{A g}(z) = B(1 - z)\,\tilde g(z - 1)$, wherever the identity defining $h$ holds
pointwise. Every application below \emph{exhibits} its $h$ rather than inferring it --- for the
profiles $g = H(s\,\cdot)$ it is $h(x) = s\,x\,H(sx)$, \ref{lem:profile-eigenfunction} --- so the
cited inversion theorem is available but never spent. Note finally the covariant scaling: $A$
commutes with dilations up to the weight $x^{-1}$, i.e.\ it scales like $\partial_t$ under
$(t, x) \mapsto (\sigma t, \sigma x)$, as (A8) requires of any inversion.
\statusA\quad\ledger{A12}\quad\emph{(Mellin inversion on a line of absolute convergence;
\sscite{widder1941laplace}, Ch.~VI, \S9, Thm.~9a. Cited, not reproven here. The operator itself is
\textbf{machine-checked}, as \texttt{Hemigroup.SelfDecomposableExponent.inversionOperator}: the
contour integral is Mathlib's \texttt{mellinInv}, since parametrising the line by $z = c + iy$
turns $dz$ into $i\,dy$ and cancels the $i$. It is defined for \emph{every} $g$, the hypotheses
having moved onto the theorems that compute it.
\textbf{Assignment.} Narrowed 2026-08-12, when the node was formalised. \ledger{A12} now
carries one step and not three: that absolute integrability of $B(-z)\tilde g(z)$ on the line
produces the function $B(\theta)g$. Everything after that function exists is \textbf{[T]} and is
proved --- \ref{lem:inversion-operator-action}.}

\emph{The narrowing came out of writing the definition
down. ``$B(\theta)g$'' is a functional calculus for an operator with poles, so the second display
is not a rewriting of the first but an assertion that a certain function \emph{exists}; Mathlib's
inversion theorem recovers a function from its own transform and says nothing about the integral
of a product, which is exactly the same gap seen from the other side. So the formal statement
takes the referent as given --- the $h$ with $\tilde h = B(-z)\tilde g$ on the line --- and what
is left over is its production, which is the cited theorem and nothing else. Whether \ledger{A12}
retires is therefore now a question about \emph{this article}: does every use of $A$ exhibit its
$h$? Each use here is $A$ applied to a profile $H(s\,\cdot)$, where $h(x) = s\,x\,H(sx)$ is
explicit and $\tilde h(z) = s^{-z}\tilde H(z+1)$ is \ref{lem:inversion-symbol}'s recursion with
the denominator cleared.}

\emph{The ledger records a correction to the draft here: it cited \sscite{samko1993fractional} for

\end{definition}

\begin{lemma}[uniqueness of the symbol]\label{lem:symbol-uniqueness}

\statusT\quad\emph{(\textbf{Machine-checked}, both halves. Step~2 is
\ref{lem:symbol-rigidity}; step~1 is

\emph{\textbf{``Of the form $x^{-1}B(\theta)$'' is the hypothesis, and reading it that way is what

\ref{def:inversion-operator}'s transform identity, hence the \emph{production} of $B(\theta)g$ from

the class quantified over in the statement is exactly the class for which the referent exists. So

\emph{\textbf{Two things the proof does not need.} It needs no injectivity of the inverse Mellin
transform --- the obvious tool, and a second citation (\sscite{widder1941laplace}, Thm.~6a, listed

looks.)}
\end{lemma}

\begin{proof}
\leanok
\uses{lem:symbol-rigidity,lem:profile-eigenfunction}
Both operators send $H(s\,\cdot)$ to $s\,H(s\,\cdot)$, so on $(0,\infty)$ their realising functions
agree; call the common value $h$. Then $B_1(-z)\,\tilde g(z) = \tilde h(z) = B_2(-z)\,\tilde g(z)$ at
every point of the line where the identity holds, and $\tilde g(z) = s^{-z}\tilde H(z)$, so cancelling
the nonvanishing factor $s^{-z}$ leaves the hypothesis of \ref{lem:symbol-rigidity}, which
concludes. At a zero of $\tilde H$ both sides of that hypothesis vanish, so it holds on the strip
outright.
\end{proof}

\begin{lemma}[the memory line stores fractional integrals]\label{lem:memory-fractional-integrals}
\uses{def:standing-hypothesis,lem:mellin-data,thm:main-characterization,lem:delay-core,lem:delayed-average-mellin}
\lean{Hemigroup.SelfDecomposableExponent.mellin_delayedField_deriv}\leanok
Under (H), let $f \in \Xz$ be causal and $u(\cdot, x) := \Phi_{0,x} f$. Then for every $t > 0$
and $1 < \operatorname{Re} z < z_*$,
\[
  \widetilde{u(t, \cdot)}(z) \;=\; \int_0^\infty x^{z-1}\, u(t, x)\,dx
  \;=\; \tilde H(z)\,\bigl(I^{z} f\bigr)(t),
\]
absolutely convergent. If moreover $f \in \core$, the same holds for $\partial_t u$ --- the
$\Xz$-derivative of \S10, i.e.\ $u(\cdot, x)$ is the primitive of $\Ex[f'(\cdot - xT_1)]$ --- with
$I^z f$ replaced by $I^z f' = I^{z-1} f$; and in that case the identity extends to the wider strip
$0 < \operatorname{Re} z < z_*$, since $f \in \core$ is bounded ($f(0) = 0$ and $f' \in \Xz$ give
$\|f\|_\infty \le \|f'\|_1$) and the weight $y^{z-1}$ is then integrable at the origin on its own.
\statusT
\end{lemma}

\begin{proof}
\leanok
In the canonical gauge $u(t, x) = \Ex\bigl[f(t - x T_1)\bigr]$. Tonelli applies, since for
$c = \operatorname{Re} z > 1$,
$\Ex[T_1^{-c}]\int_0^t y^{c-1}|f(t-y)|\,dy \le \Ex[T_1^{-c}]\, t^{c-1}\|f\|_1 < \infty$, and
gives
\[
  \int_0^\infty x^{z-1}\,\Ex\bigl[f(t - xT_1)\bigr]\,dx
  \;=\; \Ex\bigl[T_1^{-z}\bigr]\int_0^t y^{z-1} f(t - y)\,dy
  \;=\; \Ex\bigl[T_1^{-z}\bigr]\,\Gamma(z)\,(I^z f)(t),
\]
which is the claim by \ref{lem:mellin-data}. For $f \in \core$: $u(\cdot,x) = \mu_{0,x} * f$ and
$f = \mathbf{1}_{[0,\infty)} * f'$, so $u(\cdot,x) = \mathbf{1}_{[0,\infty)} * (\mu_{0,x} * f')$ ---
that is, $u(\cdot,x)$ is the primitive of $\Ex[f'(\cdot - xT_1)]$, which is what
$\partial_t u = \Ex[f'(\cdot - xT_1)]$ asserts in $\Xz$. Finally $I^zf' = I^{z-1}f$: substituting
$f(r) = \int_0^r f'$ and exchanging the order of integration over $0 < \rho \le r \le t$ leaves
$\int_\rho^t (t-r)^{z-2}dr = (t-\rho)^{z-1}/(z-1)$, and $\Gamma(z) = (z-1)\Gamma(z-1)$; the
semigroup identity $I^{z-1}I^1 = I^z$ is not needed (\ref{lem:fractional-integral-derivative}).
\end{proof}

\begin{theorem}[Theorem 4$'$: the signaling form]\label{thm:signaling-form}
\uses{def:standing-hypothesis,lem:mellin-data,lem:mellin-vertical,lem:inversion-symbol,def:inversion-operator,lem:profile-eigenfunction,lem:symbol-uniqueness,lem:memory-fractional-integrals,def:cascade-family,thm:main-characterization}
Assume (H) and work in the canonical gauge. Then:
\begin{enumerate}
\item \emph{(Eigenfunctions.)} For every $s > 0$ the profile $H(s\,\cdot)$ lies in the domain
of \ref{def:inversion-operator} for every $c \in (0, z_*-1)$, and
\[
  A\bigl[H(s\,\cdot)\bigr](x) \;=\; s\, H(sx), \qquad x > 0 .
\]
\item \emph{(The field solves the signaling problem.)} Let $f \in \core$ and
$u(\cdot, x) := \Phi_{0,x} f$. Then $u$ is causal in $t$ with $u(\cdot, x) \to f$ in $\Xz$ as
$x \downarrow 0$, and
\begin{itemize}
\item Laplace form: for every $s > 0$, $A\bigl[\hat u(s, \cdot)\bigr] = s\,\hat u(s, \cdot)$,
with $\hat u(s, \zp) = \hat f(s)$;
\item time-domain (Mellin) form: for every $t > 0$ and $1 < \operatorname{Re} z < z_*$,
\[
  \widetilde{\partial_t u(t, \cdot)}(z) \;=\; B(1 - z)\;\widetilde{u(t, \cdot)}(z - 1),
\]
that is, $\partial_t u = A\,u$ read through the Mellin transform in $x$.
\end{itemize}
\item \emph{(Uniqueness.)} $A$ is the unique operator in the covariant Mellin class with
property (1), hence with property (2).
\end{enumerate}
\statusT\quad\emph{(the hemigroup analogue of \sscite{fagerstrom2005temporal}, Thm.~4, whose
translation-invariant operator is recovered in \ref{ex:stable-signaling} as an accident of
homogeneity. Proved here, not cited.}

\texttt{Hemigroup.SelfDecomposableExponent.mellin\_signaling\_form}; what is left of (2) is its

\begin{proof}
(1) The realising function is exhibited: put $h(x) := s\,x\,H(sx)$, so that
$\tilde h(z) = s^{-z}\tilde H(z+1)$, the weight $x$ shifting the Mellin argument by one and the two
powers of $s$ combining as $s\cdot s^{-(z+1)}$. On the other side
$\tilde g(z) := \widetilde{H(s\,\cdot)}(z) = s^{-z}\tilde H(z)$, so
\[
  B(-z)\,\tilde g(z)
  = \frac{\tilde H(z+1)}{\tilde H(z)}\cdot s^{-z}\tilde H(z)
  = s^{-z}\tilde H(z+1)
  = \tilde h(z)
\]
at every point of the line where $\tilde H(z) \ne 0$, which is all but an isolated --- hence null
--- set of it. The cancellation is the whole content, being \ref{lem:inversion-symbol}'s recursion
with the denominator cleared; it should \emph{not} be described as a product containing no
division, since the division is there and clearing it is what costs the exceptional set. Both
convergence hypotheses of \ref{def:inversion-operator} hold for $h$ at height $c$, being
\ref{lem:mellin-data} and \ref{lem:mellin-vertical} at height $c + 1 \in (1, z_*)$, and $h$ is
continuous because $H$ is. So $h$ realises $B(\theta)H(s\,\cdot)$ and
\ref{lem:inversion-operator-action} gives $A[H(s\cdot)](x) = h(x)/x = s\,H(sx)$: the weight
$x^{-1}$ of \ref{def:inversion-operator} is exactly what cancels the weight $x$ that shifted the
transform, so the exhibition of $h$ and the eigenfunction relation are one statement rather than
two. This clause is \ref{lem:profile-eigenfunction}.

(2) Causality and the boundary attainment are (A3) and (A7). The Laplace form is (1) applied
to $\hat u(s, \cdot) = \hat f(s) H(s\,\cdot)$. The Mellin form combines
\ref{lem:memory-fractional-integrals} with \ref{lem:mellin-data}:
\[
  B(1-z)\,\widetilde{u(t,\cdot)}(z-1)
  = \frac{\tilde H(z)}{\tilde H(z-1)}\cdot\tilde H(z-1)\,(I^{z-1}f)(t)
  = \tilde H(z)\,(I^{z-1}f)(t)
  = \widetilde{\partial_t u(t,\cdot)}(z).
\]

(3) is \ref{lem:symbol-uniqueness}.
\end{proof}

\begin{remark}[classical interpretation; poles]\label{rem:poles}
\uses{thm:signaling-form,lem:mellin-data,lem:inversion-symbol}
The time-domain equation in \ref{thm:signaling-form}(2) is stated at the level of Mellin
transforms, and that is the honest general formulation: $B$ may have poles, at zeros of
$\tilde H$, and polynomial growth along vertical lines, so $A u(t, \cdot)$ need not be given by
an absolutely convergent integral for rough $f$. On the field the identity is nevertheless
regular, the factor $\tilde H(z-1)$ cancelling every denominator.

When the symbol $B$ is a \emph{polynomial}, $A$ is a local differential operator and the
equation is a classical PDE. That is precisely the situation of Theorem~5$'$ in Chapter~12,
and of \ref{ex:bessel-quadratic} below.
\end{remark}

\begin{example}[the Gamma family; scale-recursive form]\label{ex:gamma-signaling}
\uses{prop:gamma-family,ex:gamma-recursive,def:inversion-operator}
For $F(s) = \gamma \log(1+s)$ with $\gamma > 1$ we have
$\tilde H(z) = \Gamma(z)\Gamma(\gamma - z)/\Gamma(\gamma)$ and $B(-z) = z/(\gamma - 1 - z)$,
i.e.
\[
  B(\theta) = -\,\theta\,\bigl(\theta + \gamma - 1\bigr)^{-1},
  \qquad A = -\,\frac{1}{x}\,\theta\,\bigl(\theta + \gamma - 1\bigr)^{-1},
\]
a \emph{rational} Mellin symbol --- the memory-line mirror of the time-side resolvent
$\gamma\,\partial_t(1 + x\partial_t)^{-1}$ of \ref{ex:gamma-recursive}. Explicitly,
$w := (\theta + \gamma - 1)^{-1} u$ solves $x\,w_x + (\gamma-1)w = u$, with the regular-at-$0$
solution
\[
  w(t, x) = x^{-(\gamma-1)}\int_0^x y^{\gamma - 2}\, u(t, y)\,dy,
\]
so the time evolution at scale $x$ is computed from the field at \emph{finer} scales by a
single weighted average --- scale-recursive, one-sided from the boundary, matching the causal
architecture of the time side.
\end{example}

\begin{example}[parabolic gauge; the Bessel family is the quadratic case]\label{ex:bessel-quadratic}
\uses{prop:bessel-family,rem:gauge-freedom,def:inversion-operator,rem:poles}

$\alpha^{-1}\theta$, and the inversion becomes $A = x^{-1/\alpha}\,B(\alpha\,\theta)$. Take the
parabolic gauge $\alpha = \tfrac12$ and the Bessel-K family of \ref{prop:bessel-family},

  B(-z) = 2z(z + a), \qquad

$\partial_t u = \tfrac12 u_{xx} + \tfrac{\beta}{x}u_x$ with $u(t,0) = f(t)$ and $u(0,x) = 0$.
\end{example}

\begin{remark}[Markov media]\label{rem:markov-media}
\uses{thm:signaling-form,ex:stable-signaling,ex:gamma-signaling}
Within the class of \ref{thm:signaling-form}, the operators $A$ that additionally satisfy the
positive maximum principle --- i.e.\ generate a sub-Markov process on the memory half-line, so
that the medium is a physical transport medium --- are, after the logarithmic substitution
$x = e^\xi$, exactly those whose symbol $B$ is the Laplace exponent of a Lévy process. By the
Lamperti correspondence (\sscite{lamperti1972semi}) these media are positive self-similar
Markov processes absorbed at the boundary, and the kernels are the boundary-arrival times
$T_x \stackrel{d}{=}$ absorption time from depth $x$.

The spectral orientation of the jumps is \emph{not} constrained. For spectrally positive media
the process creeps downward through every level and the hemigroup (A6) holds \emph{pathwise},
the kernels being genuine level-passage times; but media with downward jumps are also
admissible, the factorization then holding in law though not pathwise, being inherited from
the exponent $F$ rather than from the paths.

The Gamma family is the standard example: unfolding \ref{ex:gamma-signaling}, its medium is
the multiplicative jump process that waits at scale $x$ an $\mathrm{Exp}(\text{mean } x)$ time
and jumps to $\sigma x$ with $\sigma \sim \mathrm{Beta}(\gamma-1, 1)$, reaching the boundary by
accumulation of jumps; a first-jump computation confirms
$\Ex_x[e^{-s\tau_0}] = (1+sx)^{-\gamma}$.

A necessary condition in all cases is quadratic symbol growth, $B(\theta) = O(\theta^2)$; by
\ref{ex:stable-signaling} the homogeneous members with $\alpha < \tfrac12$, of symbol order
$1/\alpha > 2$, admit \emph{no} Markov medium, although \ref{thm:signaling-form} always
supplies the analytic signaling form --- the operator exists, the process does not. The
classification of which admissible $F$ are Markov-realizable is the memory-line inverse
problem, and is taken up with Theorem~5$'$.
\end{remark}

$\widetilde{H(s\cdot)}(z) = s^{-z}\tilde H(z)$ and $\widetilde{s\,x\,H(sx)}(z) = s^{-z}\tilde H(z+1)$, the
second because the weight $x$ shifts the Mellin argument by one. Their ratio is $B(-z)$ wherever
$\tilde H(z) \ne 0$, which is almost every point of the line. The transform of $h$ converges there
by 
ef{lem:mellin-data} and its inversion integral by 
ef{lem:mellin-vertical}, both at height
$c+1 \in (1, z_*)$, so 
ef{lem:inversion-operator-action} applies and gives
$A[H(s\cdot)](x) = h(x)/x = s\,H(sx)$.
\end{proof}

\begin{lemma}[the delayed average is the field, and its transform]\label{lem:delayed-average-mellin}
\uses{def:standing-hypothesis,lem:mellin-data,prop:canonical-gauge,def:cascade-family}
\lean{Hemigroup.SelfDecomposableExponent.mellin_delayedField}\leanok
Assume (H) and work in the canonical gauge; let $f \in L^1$ be causal and $t > 0$. Then for every
$x > 0$,
\[
  \bigl(\Phi_{0,x}f\bigr)(\cdot) \;=\; \Ex\bigl[f(\,\cdot - xT_1)\bigr] \quad\text{a.e.},
\]
and for $1 < \operatorname{Re} z < z_*$ the right-hand side, read as a function of $x$, has
\[
  \int_0^\infty x^{z-1}\,\Ex\bigl[f(t - xT_1)\bigr]\,dx \;=\; \tilde H(z)\,\bigl(I^z f\bigr)(t),
\]
both sides absolutely convergent.
\statusT\quad\emph{(\textbf{Machine-checked}, as
\texttt{Hemigroup.SelfDecomposableExponent.mellin\_delayedField}, with
\texttt{coeFn\_Phi\_zero} (the first display), \texttt{kernel\_zero\_eq\_map\_lawT}$_1$ (the
canonical gauge at the level of measures), \texttt{integrable\_delayed} (the Fubini side
condition) and \texttt{mul\_riemannLiouville} (the reflection);
\texttt{\#print axioms} gives \ledger{A17} and nothing else. This is the first clause of
\ref{lem:memory-fractional-integrals}; what is left there is its derivative clause.}

\emph{\textbf{The quantifiers do not commute, and that is the honest reason a representative has to
be chosen.} $\Phi_{0,x}$ acts on $L^1$ classes, so the first display holds for each $x$ only up to
a null set of times --- and the null set depends on $x$. ``For each $x$, for a.e.\ $t$'' does not
give ``for a.e.\ $t$, for every $x$'', so the second display, which is an integral over $x$ at a
fixed $t$, cannot be taken of $(\Phi_{0,x}f)(t)$ as it stands. Some representative must be named
whichever way the identification is read. What the choice made here buys is that it is
\emph{local}: one function, $\Ex[f(t-xT_1)]$, defined outright, with the first display as its
bridge --- rather than a pointwise model of $\Phi$ built under all of Chapters 10--12.}

\emph{\textbf{$I^z$ is defined in the development, not cited.} Mathlib carries no fractional
integral of any order, so the Riemann--Liouville integral is given its definition here. That adds
no interface: \sscite{samko1993fractional} is cited in this chapter for the notation and the
theory of $I^z$, and nothing here uses more of that theory than the definition itself.}

\emph{\textbf{The exchange is licensed by the two ends of the strip, one apiece.} Inside, the
dilate of the past integrand is integrable because $\operatorname{Re} z > 1$: on $(0,t]$ the
weight $y^{c-1}$ is bounded by $t^{c-1}$, so $f$ needs nothing beyond $L^1$, and beyond $t$
causality kills the integrand outright. Outside, the dilation $x \mapsto \tau x$ leaves exactly
the factor $\tau^{-z}$, whose expectation is finite because $\operatorname{Re} z < z_*$. That is
the same shape as \ref{lem:mellin-data}'s hinge with the lower end moved from $0$ to $1$ --- and
the move is what \emph{makes} (H) ask for $z_* > 1$ rather than $z_* > 0$. Below $1$ this lemma
has no strip to live in and \ref{thm:signaling-form}(2) would have nothing to say, which is also
why \ref{prop:extreme-rays}'s observation that every extreme ray has $z_* = 1$ bites here.)}
\end{lemma}

\begin{proof}
\leanok
\uses{lem:mellin-data}
Everything factors through the past integral $\int_0^\infty y^{z-1}f(t-y)\,dy$. Fubini exchanges
the two integrals; the inner one is then a dilate, contributing $\tau^{-z}$ times the past
integral, and the outer integral of $\tau^{-z}$ is $\Ex[T_1^{-z}]$. The past integral is
$\Gamma(z)(I^zf)(t)$ by the reflection $y \mapsto t-y$, whose domains match because causality has
already confined the integral to $(0,t]$; and $\Ex[T_1^{-z}]\Gamma(z) = \tilde H(z)$ by
\ref{lem:mellin-data}. The two $\Gamma$'s cancel, which is why the statement has none.
\end{proof}

\begin{lemma}[the Mellin form of the signaling equation, transform side]\label{lem:signaling-mellin-form}
\uses{lem:delayed-average-mellin,lem:inversion-symbol,lem:memory-fractional-integrals}
\lean{Hemigroup.SelfDecomposableExponent.inversionSymbol_mul_mellin_delayedField}\leanok
Assume (H), canonical gauge; let $f \in \core$ be causal, $t > 0$ and
$1 < \operatorname{Re} z < z_*$, and suppose $\tilde H(z-1) \ne 0$. Then
\[
  B(1-z)\;\widetilde{u(t,\cdot)}(z-1) \;=\; \tilde H(z)\,\bigl(I^{z-1}f\bigr)(t).
\]
\statusT\quad\emph{(\textbf{Machine-checked}, as
\texttt{Hemigroup.SelfDecomposableExponent.inversionSymbol\_mul\_mellin\_delayedField};
\texttt{\#print axioms} gives \ledger{A17} and nothing else. This is the displayed computation in
the proof of \ref{thm:signaling-form}(2), entire; what stands between it and that node's
conclusion is the identification of the right-hand side with
$\widetilde{\partial_t u(t,\cdot)}(z)$, which is the derivative clause of
\ref{lem:memory-fractional-integrals} and needs \ref{lem:delay-core}.}

\emph{\textbf{Two hypotheses that the prose does not carry, for the same two reasons the chapter
has met before.} $\operatorname{Re} z > 1$ is used at $z-1$, where the strip must start at $0$
rather than $1$ --- so the boundedness of $f \in \core$ is load-bearing here and not incidental;
see \ref{lem:memory-fractional-integrals}. And $\tilde H(z-1) \ne 0$ is needed because $B$ is a
quotient: \ref{rem:poles} says the identity ``is nevertheless regular, the factor $\tilde H(z-1)$
cancelling every denominator'', which is true of the product as a \emph{meromorphic function} and
not of $B$ evaluated at a point, where the division is present and has to be cancelled. That is
the third time in this chapter that a symbolic cancellation has had to become a pointwise one,
after \ref{thm:signaling-form}(1) and \ref{lem:symbol-rigidity}.)}
\end{lemma}

\begin{proof}
\leanok
\uses{lem:delayed-average-mellin,lem:inversion-symbol}
$B(1-z) = \tilde H(z)/\tilde H(z-1)$ by \ref{lem:inversion-symbol}, and
$\widetilde{u(t,\cdot)}(z-1) = \tilde H(z-1)(I^{z-1}f)(t)$ by \ref{lem:delayed-average-mellin} at
$z-1$, whose strip condition $0 < \operatorname{Re}(z-1) < z_*$ is the hypothesis. Cancel
$\tilde H(z-1)$.
\end{proof}

\begin{lemma}[the fractional integral absorbs a derivative]\label{lem:fractional-integral-derivative}
\uses{lem:memory-fractional-integrals}
\lean{Hemigroup.riemannLiouville_integral}\leanok
Let $\operatorname{Re} z > 1$, let $g$ be integrable on $(0,t]$ and set
$f(r) := \int_0^r g$ --- that is, $f$ absolutely continuous with $f(0) = 0$ and $f' = g$. Then
\[
  \bigl(I^{z} g\bigr)(t) \;=\; \bigl(I^{z-1} f\bigr)(t).
\]
\statusT\quad\emph{(\textbf{Machine-checked}, as \texttt{Hemigroup.riemannLiouville\_integral},
with \texttt{integral\_cpow\_sub\_left} for the inner integral; \texttt{\#print axioms} reduces to
Lean core alone --- this statement is about the Riemann--Liouville family and mentions no
hemigroup object, so not even \ledger{A17} enters.}

\emph{\textbf{The semigroup property is not needed, and not needing it is cheaper.} The draft
derives this as $I^zf' = I^{z-1}I^1f' = I^{z-1}f$, i.e.\ from $I^{z-1}I^1 = I^z$ --- a Beta-integral
identity that would have to be proved in its own right, Mathlib having no fractional integral at
all. Substituting $f(r) = \int_0^r g$ and exchanging the order of integration over the triangle
$0 < \rho \le r \le t$ instead leaves the inner integral
$\int_\rho^t (t-r)^{z-2}dr = (t-\rho)^{z-1}/(z-1)$, after which $\Gamma(z) = (z-1)\Gamma(z-1)$
finishes it. The condition $\operatorname{Re} z > 1$ is exactly what makes that inner integral
converge --- the same endpoint that \ref{lem:delayed-average-mellin} and
\ref{lem:signaling-mellin-form} turn on, arrived at for a third independent reason.)}
\end{lemma}

\begin{proof}
\leanok
Write $f(r) = \int_0^r g$ and exchange the order of integration in
$\Gamma(z-1)(I^{z-1}f)(t) = \int_0^t (t-r)^{z-2}\int_0^r g(\rho)\,d\rho\,dr$ over the triangle
$0 < \rho \le r \le t$, which is legitimate because the same computation on absolute values is
bounded by $t^{z-1}\|g\|_1/(\operatorname{Re} z - 1)$. The inner integral is
$\int_\rho^t (t-r)^{z-2}dr = (t-\rho)^{z-1}/(z-1)$, so the double integral is
$(z-1)^{-1}\Gamma(z)(I^zg)(t)$; and $\Gamma(z) = (z-1)\Gamma(z-1)$.
\end{proof}
