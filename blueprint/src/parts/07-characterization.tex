% draft: Lemma 7.1
\begin{lemma}[self-decomposable exponents]\label{lem:selfdecomposable-exponents}
\uses{def:bernstein-function,prop:bernstein-toolbox}
For $F \in \BFz$ the following are equivalent:
\begin{enumerate}
\item $F(b\,\cdot) - F(a\,\cdot) \in \BFz$ for all $0 < a \le b$;
\item $B(s) := s\,F'(s)$ is a Bernstein function;
\item $F$ has the representation
\[
  F(s) = b_0\, s + \int_0^\infty \bigl(1 - e^{-st}\bigr)\, \frac{k(t)}{t}\,dt,
  \qquad b_0 \ge 0, \tag{7.1}
\]
with $k : (0,\infty) \to [0,\infty)$ nonincreasing, $\int_0^1 k < \infty$ and
$\int_1^\infty k(t)/t\,dt < \infty$.
\end{enumerate}
\statusT\quad\emph{(a collation, like \ref{thm:main-characterization} and for the same reason:
its implications are proved by different means, at different cost, and one of them is
formalised. The cycle is $(1) \Rightarrow (2) \Rightarrow (3) \Rightarrow (1)$, the first two
steps being \ref{lem:selfdecomposable-derivative} and the last
\ref{lem:selfdecomposable-increment}. \textbf{The asymmetry is the point.}
$(3) \Rightarrow (1)$ is elementary from the representation and is machine-checked;
$(1) \Rightarrow (2)$ and $(2) \Rightarrow (3)$ rest on \ref{prop:bernstein-toolbox}(4) and on
the uniqueness clause of \ref{prop:bernstein-toolbox}(3) --- \ledger{A4} and \ledger{A3} --- so
formalising them means formalising those, which is what the trust boundary declines to do. This
node is therefore not marked as proved in Lean; its expensive half is instead carried as
\ledger{A18}, which \ref{thm:main-analysis} spends and \ref{thm:main-construction} does not.
What a consumer needs is almost always one of the two halves, and citing the right one is what
keeps the construction off \ledger{A3}, \ledger{A4} and \ledger{A18} alike --- a separation
\texttt{CIAxiomGuard.lean} now checks per node.)}
\end{lemma}

\begin{proof}
$(1) \Rightarrow (2)$ and $(2) \Rightarrow (3)$ are \ref{lem:selfdecomposable-derivative};
$(3) \Rightarrow (1)$ is \ref{lem:selfdecomposable-increment}. The cycle gives the three
equivalences.
\end{proof}

% draft: Remark 7.2
\begin{remark}[the probabilistic reading]\label{rem:sato-reading}
\uses{lem:selfdecomposable-exponents}
Condition (3) is the classical characterization of the Laplace exponents of
\emph{self-decomposable} laws on $[0,\infty)$, the laws of $L$-class; see
\sscite{sato1999levy}, Ch.~3. Probabilistically, (7.1) together with the similarity structure
below says that the kernels are the marginal laws of a \emph{self-similar additive process},
a Sato subordinator: an increasing process $(T_{\tilde x})_{\tilde x \ge 0}$ with independent
--- but not stationary --- increments, satisfying
$T_{\sigma \tilde x} \stackrel{d}{=} \sigma\, T_{\tilde x}$.
\end{remark}

% draft: Theorem 7.3
\begin{theorem}[Main theorem]\label{thm:main-characterization}
\uses{def:cascade-family,thm:main-construction,thm:main-analysis,prop:main-uniqueness}
A family $(\Phi_{x,y})_{0 \le x \le y}$ satisfies (A1)--(A8) and (ND) \textbf{iff} there exist
an increasing bijection $\chi$ of $[0,\infty)$ --- the gauge --- and a function $F$ of the
form (7.1) with $F \not\equiv 0$ such that
\[
  \Phi_{x,y} f = \mu_{x,y} * f, \qquad
  \hat\mu_{x,y}(s) = \exp\Bigl[-\bigl(F(\chi(y)\,s) - F(\chi(x)\,s)\bigr)\Bigr].
\]
In particular, in the canonical gauge the scale-space kernels are
\[
  \phi_{x}(t) \;=\; \Lap^{-1}\Bigl[\, e^{-F(x\,s)} \,\Bigr](t),
  \qquad F \text{ as in } (7.1),
\]
and the pair $(\chi, F)$ is unique up to the normalization $\chi(1) = 1$.
\statusT\quad\emph{(the variant of \sscite{fagerstrom2005temporal}, Thm.~2, for the hemigroup
axioms. Proved here, not cited: the 2005 theorem is the semigroup statement, recovered below
as \ref{cor:semigroup-case}. \textbf{Formalisation state.} This node is a collation, and its
three halves are \ref{thm:main-construction}, \ref{thm:main-analysis} and
\ref{prop:main-uniqueness}, where the formalisation state of each is recorded. All three
are now machine-checked; the node itself still carries no tag, because it is a collation and an
equivalence is established by its parts rather than by a statement of its own. That is the whole
reason the halves are stated separately --- see the note at the head of this file. What the
separation buys is visible in the ledger: $(\Leftarrow)$ and the uniqueness clause rest on
\ledger{A17}, $(\Rightarrow)$ on \ledger{A18}, and neither borrows the other's.)}
\end{theorem}

\begin{proof}
$(\Rightarrow)$ is \ref{thm:main-analysis}, $(\Leftarrow)$ is \ref{thm:main-construction}, and
the uniqueness clause is \ref{prop:main-uniqueness}.
\end{proof}

% draft: Corollary 7.4
\begin{corollary}[recovering the semigroup case]\label{cor:semigroup-case}
\uses{thm:main-characterization,lem:covariance-laplace,prop:canonical-gauge}
Assume in addition that the family is one-parameter: $\Phi_{x,y}$ depends only on $y - x$.
Then, after the normalization $g_{0,1}(1) = 1$,
\[
  F(s) = s^{\alpha} \quad \text{for some } 0 < \alpha \le 1,
\]
i.e.\ the kernels are exactly the extremal stable densities of
\sscite{fagerstrom2005temporal}, Thm.~2, together with the boundary case $\alpha = 1$ --- the
pure delay $\mu_{0,x} = \delta_x$.
\statusT
\end{corollary}

\begin{proof}
Homogeneity gives $g_{x,y} = (y - x)\,g$ with $g := g_{0,1}$, and (6.1) becomes
$(\Sact_\sigma y - \Sact_\sigma x)\, g(s) = (y - x)\, g(\sigma s)$. Hence $\Sact_\sigma$ is
affine, and fixing $0$ it is linear: $\Sact_\sigma x = c(\sigma) x$ with
$c(\sigma) g(s) = g(\sigma s)$.

Then $c$ is a continuous multiplicative homomorphism of $(0,\infty)$, so $c(\sigma) =
\sigma^\alpha$ by Cauchy's functional equation, as in \sscite{fagerstrom2005temporal},
Lemma~2. Now $g(\sigma s) = \sigma^\alpha g(s)$ forces $g(s) = g(1) s^\alpha$, and the
normalization makes $g(1) = 1$. Membership of $\BFz$ restricts $\alpha$ to $(0,1]$. The value
$\alpha = 1$ gives $sF'(s) = s \in \BF$, so \ref{lem:selfdecomposable-exponents} admits it and
it is not excluded by the present axioms.
\end{proof}

% draft: Remark 7.5
\begin{remark}[the boundary case $\alpha = 1$ and the drift term]\label{rem:drift-boundary}
\uses{cor:semigroup-case,lem:selfdecomposable-exponents}
\sscite{fagerstrom2005temporal} excluded $\alpha = 1$ through the pointwise-continuity
requirement on the kernel functions, which the deterministic delay $\delta(t - x)$ violates.
Under \ref{def:cascade-family} the delay family --- and more generally the drift term $b_0 s$
in (7.1), which shifts every kernel by $b_0 x$ --- is admissible.

If that is undesired it is removed by one additional axiom: absolute continuity of the kernels
for $x > 0$, or the requirement $\lim_{s\to\infty} F(s)/s = 0$, which says there is no
deterministic front.
\end{remark}

\begin{proposition}[Choquet structure of the admissible cone]\label{prop:extreme-rays}
\uses{thm:main-characterization,lem:selfdecomposable-derivative,prop:bernstein-toolbox}
The admissible exponents of \ref{thm:main-characterization} form a convex cone, and the map
\[
  (b_0, \rho) \;\mapsto\; F, \qquad
  F(s) = b_0\,s + \int_{(0,\infty)} \mathrm{Ein}(\tau s)\,\rho(d\tau),
  \qquad \mathrm{Ein}(z) := \int_0^z \frac{1 - e^{-u}}{u}\,du,
\]
is a linear bijection from
$\bigl\{(b_0, \rho) : b_0 \ge 0,\ \rho \in \Mplus(0,\infty),\
\int \bigl[\tau \wedge (1 + \log_+ \tau)\bigr]\,\rho(d\tau) < \infty \bigr\}$
onto that cone, with inverse $\rho = -dk$, $k(t) = \rho\bigl((t,\infty)\bigr)$. Consequently the
extreme rays of the cone are the pure delay $s \mapsto s$ together with the one-parameter family
$\mathrm{Ein}(\tau\,\cdot)$, $\tau > 0$ --- the Laplace exponents, with L\'evy density
$t^{-1}\mathbf{1}_{(0,\tau)}(t)$, of the \emph{dilated Dickman subordinators}
(\sscite{caravenna2019dickman}, (1.1)--(1.2), p.~1) --- and every admissible family is a unique
superposition of Dickman rays and drift.
\statusT\quad\emph{(The integrability condition is not in the draft's Remark 7.6; it is forced
by the bijection, being exactly finiteness of $F$, and is written out here. Not formalised: it
needs Lebesgue--Stieltjes measures and a Choquet argument, neither of which the
$\Rightarrow$ direction
consumes.)}
\end{proposition}

\begin{proof}
By \ref{thm:main-characterization} an admissible $F$ is exactly one of the form (7.1) with $b_0
\ge 0$ and $k$ nonincreasing, and $k$ may be taken right-continuous with $k(\infty) = 0$, the
latter forced by $\int^\infty k(t)\,t^{-1}dt < \infty$. So $\rho := -dk$ is a positive measure
on $(0,\infty)$ with $k(t) = \rho((t,\infty))$, and Tonelli gives
\[
  \int_0^\infty \bigl(1 - e^{-st}\bigr)\,\frac{k(t)}{t}\,dt
  = \int_{(0,\infty)} \rho(d\tau) \int_0^\tau \bigl(1 - e^{-st}\bigr)\,\frac{dt}{t}
  = \int_{(0,\infty)} \mathrm{Ein}(\tau s)\,\rho(d\tau),
\]
which is the displayed representation. Since $\mathrm{Ein}(z) \le z$ and
$\mathrm{Ein}(z) = \log z + \gamma_E + o(1)$ as $z \to \infty$, the integral is finite for one
$s > 0$ iff it is finite for all, iff $\int [\tau \wedge (1 + \log_+\tau)]\,\rho(d\tau) <
\infty$; so the stated domain is exactly the set of pairs the representation accepts. The map
is linear in $(b_0, \rho)$ by construction and surjective by the above. It is injective because
the L\'evy--Khintchine triple of $F$ is unique (\ref{prop:bernstein-toolbox}(3)): the triple
determines $b_0$ and the L\'evy measure $k(t)t^{-1}dt$, hence $k$ up to a null set, hence
$\rho = -dk$.

A linear bijection carries extreme rays to extreme rays, so it suffices to identify them in the
domain, which is the product of the ray $[0,\infty)$ with a cone of positive measures. Its
extreme rays are $(1, 0)$ and $(0, \delta_\tau)$, $\tau > 0$: any $(b_0, \rho)$ decomposes
uniquely as $b_0\,(1,0) + \int (0,\delta_\tau)\,\rho(d\tau)$, and a pair is extreme only if it
admits no nontrivial such splitting, which forces $\rho = 0$ or $\rho$ a single atom with
$b_0 = 0$. Under the map, $(1,0) \mapsto s$, the pure delay; and $(0, \delta_\tau) \mapsto
\mathrm{Ein}(\tau s)$, whose $k(t) = \delta_\tau((t,\infty)) = \mathbf{1}_{(0,\tau)}(t)$ gives
the L\'evy density $t^{-1}\mathbf{1}_{(0,\tau)}(t)$.
\end{proof}

\begin{remark}[the three corners are slices, not extremes]\label{rem:extreme-rays}
\uses{prop:extreme-rays,cor:semigroup-case,prop:stable-family,prop:gamma-family,prop:bessel-family,thm:locality,def:standing-hypothesis}
None of the distinguished families of this paper is extreme; each is a slice cut by a different
principle --- the stable family by dilation symmetry (\ref{cor:semigroup-case},
\ref{prop:stable-family}), the Gamma family as the atoms of the Thorin representation
(\ref{prop:gamma-family}), the Bessel family by locality (\ref{prop:bessel-family},
\ref{thm:locality}). Since all
three have completely monotone $k$, the closed convex cone the three corners generate together
is exactly the Thorin subclass --- strictly smaller than the whole class, the difference being
the non-completely-monotone monotone memory kernels (plateaus, kinks, hard cutoffs). The
Dickman rays are the paradigm of that difference: their kernels solve the delay equation
obtained from the Volterra identity of the memory chapter with the step kernel
$k = \mathbf{1}_{[0,\tau)}$, and carry derivative singularities at the delays
$\tau, 2\tau, \dots$ --- echo signatures unattainable within the Thorin subclass.

Two universal bounds nevertheless hold across the entire cone: every kernel is absolutely
continuous (\sscite{sato1999levy}, Thm.~27.13), and every kernel is \emph{unimodal}, by
Yamazato's theorem on class $L$ laws (\sscite{yamazato1978unimodality}, Thm.~1, p.~523). The
axioms never produce multimodal kernels.

Finally, every extreme ray has $F(s) \sim \log s$ and hence $z_* = 1$: the extreme boundary of
the cone lies exactly on the boundary of \ref{def:standing-hypothesis}.
\end{remark}

\begin{theorem}[the construction: every admissible $F$ gives a family]\label{thm:main-construction}
\uses{def:cascade-family,def:levy-exponent,lem:selfdecomposable-increment,lem:vanishing,prop:bernstein-toolbox,prop:laplace-uniqueness-causal,prop:laplace-continuity-causal,lem:transform-tightness,lem:convolution-representation}
\lean{Hemigroup.SelfDecomposableExponent.cascadeFamily}\leanok
Let $F$ be of the form (7.1) with $F \not\equiv 0$. For $0 \le a \le b$ let $\mu_{a,b}$ be the
probability measure on $[0,\infty)$ with
\[
  \hat\mu_{a,b}(s) = \exp\bigl[-g_{a,b}(s)\bigr], \qquad g_{a,b}(s) := F(bs) - F(as),
\]
and set $\Phi_{a,b} f := \mu_{a,b} * f$. Then $(\Phi_{a,b})_{0 \le a \le b}$ is a causal cascade
measurement family in the sense of \ref{def:cascade-family}, with scaling action
$\Sact_\sigma x = \sigma x$.
\statusT\quad\emph{(\textbf{Machine-checked}, as
\texttt{Hemigroup.SelfDecomposableExponent.cascadeFamily} --- and at the level the theorem
states it: not ``the kernels satisfy properties we named (A1)--(A8)'' but ``the kernels
\emph{are} a causal cascade measurement family'', the Lean structure of
\ref{def:cascade-family} instantiated. \texttt{\#print axioms} reduces to Lean core plus
\ledger{A17}, which is the single crossing of the trust boundary and enters at one point only:
the existence of $\mu_{a,b}$ given its triple. That is A17 rather than the Bernstein--Widder
route the proof below takes, because the development has the Lévy triple in hand and so needs a
\emph{construction}, not a representation theorem --- see
\texttt{Formalization/Hemigroup/Interfaces.lean} for why the interface is phrased so that the
day the compound-Poisson construction is carried out it becomes a lemma and the trust boundary
returns to empty, without touching a downstream statement.
\textbf{What the proof cost, against what was expected.} The passage from exponents to measures
is pure transport, the transform carrying convolution to multiplication and dilation to
reparametrisation, and being injective on causal measures --- and that injectivity is
\ref{prop:laplace-uniqueness-causal}, proved here, so it costs nothing. The lower endpoint
$a = 0$ is not a separate construction: with the normalisation $k(0) = 0$ the increment density
$k(u/b) - k(u/a)$ at $a = 0$ is $k(u/b)$ and the drift is $b_0 b$, which is exactly the dilated
representation of $F(b\,\cdot)$ (\texttt{increment\_zero\_left}). So $G(x,s) = g_{0,x}(s)$, the
object the similarity analysis of chapters 5--6 is built on, is available.
(A1)--(A5) come from two applications of one Tonelli identity,
$\int \bigl(\int g(t-r)\,\mu(dr)\bigr)dt = \|\mu\| \int g$ --- once to estimate, giving (A1);
once to license the Bochner swap, giving (A5). (A7) was expected to cost a second interface and
did not: \ref{prop:laplace-continuity-causal} is proved rather than cited, and the tightness it
needs was always ours (\ref{lem:transform-tightness}). The bridge from weak convergence to
convergence in $\Lone$ is the $\varepsilon/3$ argument below, with the compact supplied by
tightness, the left tail by causality, and the extension from $C_c$ to $\Lone$ by density
against the contraction bound.
\textbf{Two defects the instantiation caught}, neither visible from reading
\ref{def:cascade-family}: that the family is indexed from $x = 0$ while the construction began
at $x > 0$ --- repaired at the source by the normalisation above, not by narrowing the
specification --- and that each axiom must carry its index restriction $0 \le x \le y$
explicitly, without which (A5) would be \emph{false} for the constructed family, $\mu_{x,y}$
being the zero measure off the range. This is the one place in the development where an error is
silent, a mis-stated axiom making the theorem vacuous rather than unprovable, and an instance is
the only thing that catches it.)}
\end{theorem}

\begin{proof}
Work in the canonical gauge. For $0 \le a \le b$ the function $g_{a,b} = F(b\,\cdot) - F(a\,\cdot)$
lies in $\LE$ by \ref{lem:selfdecomposable-increment}, hence in $\BFz$ by
\ref{prop:bernstein-toolbox}(3), so $e^{-g_{a,b}}$ is completely monotone
with value $1$ at $\zp$ by \ref{prop:bernstein-toolbox}(2), and by the Bernstein--Widder
criterion \ref{prop:bernstein-toolbox}(1) it is the Laplace transform of a unique probability
measure $\mu_{a,b}$ on $[0,\infty)$. Then:
\begin{itemize}
\item (A1)--(A5) hold by the converse direction of \ref{lem:convolution-representation}.
\item (A6) holds because exponents add, $g_{a,b} + g_{b,c} = g_{a,c}$, and Laplace transforms
determine measures (\ref{prop:laplace-uniqueness-causal}).
\item (A8) holds with $\Sact_\sigma x = \sigma x$, since
$\hat\mu_{\sigma a, \sigma b}(s) = \hat\mu_{a,b}(\sigma s)$.
\item (ND) holds because $F$ is strictly increasing, by \ref{lem:vanishing} applied to $F$.
\end{itemize}

For (A7): if $(a_n, b_n) \to (a,b)$ then $\hat\mu_{a_n,b_n} \to \hat\mu_{a,b}$ pointwise, by
continuity of $F$. The family is tight by \ref{lem:transform-tightness}, uniformly over
$b_n \le B$, because $F(\zp) = 0$ makes the bound's numerator $F(Bs)$ as small as wanted; so
$\mu_{a_n,b_n} \to \mu_{a,b}$ weakly by \ref{prop:laplace-continuity-causal}. Weak convergence
of probability measures implies $\mu_{a_n,b_n} * f \to \mu_{a,b} * f$ in $\Lone$ for every
$f \in X$: writing $\mu * f = \int \shift_r f\, \mu(dr)$ as an $X$-valued Bochner integral, the
map $r \mapsto \shift_r f$ is bounded and continuous, and an $\varepsilon/3$ argument on a
compact carrying mass $1 - \varepsilon$ concludes, first for compactly supported continuous $f$
and then for all of $X$ by density, the error not growing because each $\Phi_{a,b}$ is a
contraction.
\end{proof}

\begin{theorem}[the analysis: every family comes from an admissible $F$]\label{thm:main-analysis}
\uses{def:cascade-family,lem:convolution-representation,lem:transform-continuity,lem:additivity,thm:increments-bernstein,cor:strict-monotonicity,lem:covariance-laplace,lem:action-rigidity,prop:canonical-gauge,lem:selfdecomposable-derivative}
\lean{Hemigroup.CascadeCore.main_analysis}\leanok
Let $(\Phi_{x,y})_{0 \le x \le y}$ satisfy (A1)--(A8) and (ND). Then there are an increasing

\begin{proposition}[uniqueness of the gauge and the exponent]\label{prop:main-uniqueness}
\uses{thm:main-construction,prop:laplace-uniqueness-causal,lem:vanishing,cor:strict-monotonicity}
\lean{Hemigroup.SelfDecomposableExponent.gauge_and_exponent_unique}\leanok
Let $F$ be of the form (7.1) with $F \not\equiv 0$, and let $F'$ be of the same form. Suppose
$\chi : [0,\infty) \to [0,\infty)$ is nondecreasing and positive on $(0,\infty)$ with
$\chi(\zp) = 0$ and $\chi(1) = 1$, and that $\mu'_{\chi(x),\chi(y)} = \mu_{x,y}$ for all
$0 < x \le y$. Then $\chi = \Id$ and $F' = F$.
\statusT\quad\emph{(\textbf{Machine-checked}, as
\texttt{Hemigroup.SelfDecomposableExponent.gauge\_and\_exponent\_unique};
\texttt{\#print axioms} is held against \texttt{blueprint/trust-boundary.txt} by CI, with
\ledger{A17} reaching the statement through the construction of the kernels it quantifies over
and nothing else. The argument itself adds no analysis to
\ref{thm:main-construction}.
\textbf{Why $\chi$ is a bare function here.} \ref{prop:canonical-gauge} produces $\chi$ as an
increasing bijection with an inverse conjugating the scaling action into multiplication, and
none of that is needed: this clause only ever \emph{quantifies over} gauges, so $\chi$ enters
carrying exactly the three properties the argument consumes. That is a strictly stronger
statement than the bijection form, and it leaves the interface for
\ref{thm:main-analysis} --- which must \emph{build} $\chi$ --- to be designed by its consumer.
\textbf{The analytic content} is that $F$ is strictly increasing when $F \not\equiv 0$
(\texttt{exponent\_strictMono}), which is what makes the $c$ with $g_{0,x} = F(c\,\cdot)$
unique. That proof runs \ref{lem:vanishing} on the \emph{increment}: if $g_{1,c}$ vanished
anywhere it would vanish identically, giving $F(cs) = F(s)$, and iterating \emph{downwards}
sends $s_0/c^n \to 0$, where continuity at the origin forces $F \equiv 0$. The one new step is
pinning the constant, the hypothesis giving only \emph{differences} of exponents; it is run as
an $\varepsilon$ argument rather than a limit, since \texttt{exists\_exponent\_lt} is already
non-sequential.)}
\end{proposition}

\begin{proof}
Equal kernels have equal exponents, the transform being injective
(\ref{prop:laplace-uniqueness-causal}) and $\exp$ injective, so the hypothesis reads
\[
  F'(\chi(y)s) - F'(\chi(x)s) = F(ys) - F(xs), \qquad 0 < x \le y,\ s \ge 0 .
\]
Let $x \downarrow 0$. Both subtrahends vanish, because $F(\zp) = F'(\zp) = 0$ and
$\chi(\zp) = 0$, leaving $F'(\chi(y)s) = F(ys)$. At $y = 1$, with $\chi(1) = 1$, this is
$F' = F$.

Feeding that back gives $F(\chi(y)s) = F(ys)$ for all $y > 0$, $s \ge 0$. At $s = 1$, strict
monotonicity of $F$ --- which holds because $F \not\equiv 0$, by \ref{lem:vanishing} applied to
the increments as in \ref{cor:strict-monotonicity} --- forces $\chi(y) = y$.
\end{proof}

\begin{lemma}[the dilation increment of a self-decomposable exponent]\label{lem:selfdecomposable-increment}
\uses{def:levy-exponent}
\lean{Hemigroup.levyExponentD_increment}\leanok
Let $F \in \LE$ have drift $b_0 \ge 0$ and Lévy density $k(t)/t$ with $k \ge 0$ nonincreasing on
$(0,\infty)$. Then for $0 < a \le b$ and $s \ge 0$,
\[
  F(a s) + G_{a,b}(s) = F(b s),
\]
where $G_{a,b} \in \LE$ is the Lévy exponent with drift $b_0(b-a)$ and Lévy density
$\bigl(k(u/b) - k(u/a)\bigr)/u$. In particular $F(b\,\cdot) - F(a\,\cdot) \in \LE$, which is
condition~(1) of \ref{lem:selfdecomposable-exponents} in representation form.
\statusT\quad\emph{(\textbf{Machine-checked}, as
\texttt{Hemigroup.levyExponentD\_increment}, \texttt{\#print axioms} reducing to Lean core ---
no ledger entry at all, which is the whole point of separating it. \textbf{Stated additively},
$F(as) + G = F(bs)$ rather than as a difference, because that is what a cascade needs
(exponents must \emph{add} along $a \le b \le c$) and because the Lean exponents are valued in
$[0,\infty]$, where subtraction truncates; the two forms agree wherever $F(as) < \infty$.
\textbf{Why this is the cheap half.} It is a change of variables and a sign: $du/u$ is the Haar
measure of the dilation group, so a dilation moves the Lévy density and nothing else, and
nonnegativity of $k(u/b) - k(u/a)$ is exactly the hypothesis that $k$ is nonincreasing. No
closure theorem, no derivative, no uniqueness of a triple --- so none of
\ref{prop:bernstein-toolbox}. Classically one would instead read $(3) \Rightarrow (2)
\Rightarrow (1)$ off \ref{lem:selfdecomposable-derivative}, which is shorter to write and costs
\ledger{A3} and \ledger{A4}; \ref{thm:main-construction} cites this node rather than that route,
and that is why its trust base is Lean core plus \ledger{A17} alone.)}
\end{lemma}

\begin{proof}
\leanok
Write $F(cs) = b_0 c s + \int_0^\infty \bigl(1 - e^{-cst}\bigr)\,k(t)\,\frac{dt}{t}$ for $c > 0$
and substitute $u = ct$. The measure $dt/t$ is invariant, so
\[
  F(cs) = b_0 c\, s + \int_0^\infty \bigl(1 - e^{-su}\bigr)\, k(u/c)\,\frac{du}{u} ,
\]
i.e.\ dilating the argument by $c$ dilates the density and rescales the drift, and does nothing
else. Subtracting the identity at $c = a$ from the one at $c = b$ gives
\[
  F(bs) - F(as) = b_0 (b-a)\, s
    + \int_0^\infty \bigl(1 - e^{-su}\bigr)\,\bigl(k(u/b) - k(u/a)\bigr)\,\frac{du}{u},
\]
which is the asserted $G_{a,b}$. It lies in $\LE$: the drift $b_0(b-a)$ is nonnegative since
$a \le b$, and the density is nonnegative since $u/b \le u/a$ for $u > 0$ and $k$ is
nonincreasing. Both integrals are finite because each is bounded by $F(bs) < \infty$.
\end{proof}

\begin{lemma}[the derivative characterization]\label{lem:selfdecomposable-derivative}
\uses{def:bernstein-function,prop:bernstein-toolbox}
Let $F \in \BFz$. If $F(b\,\cdot) - F(a\,\cdot) \in \BFz$ for all $0 < a \le b$, then
$B(s) := sF'(s)$ is a Bernstein function; and if $B$ is a Bernstein function, then $F$ has the
representation (7.1) with $k$ nonincreasing. The converse of each also holds.
\statusT\quad\emph{(\textbf{Taken as \ledger{A18}, not proved} --- a review decision recorded
2026-08-10. Both directions run through \ref{prop:bernstein-toolbox}: (1)$\Rightarrow$(2) is
closure of $\BF$ under pointwise limits, \ledger{A4}, applied to a difference quotient;
(2)$\Rightarrow$(3) needs uniqueness of the Lévy--Khintchine triple, \ledger{A3}, to identify the
density it produces with $k$. Neither is a statement about the hemigroup family --- they are
facts about $\BF$ as a class of functions --- and neither can even be \emph{stated} here, since
both quantify over $\BF$ and the development has no $\CM$. So what the trust boundary carries is
the composite, in the direction \ref{thm:main-analysis} spends, phrased in $\LE$: exactly the
device \ledger{A17} uses. Unlike \ledger{A17}, expect it to be permanent --- the
(1)$\Rightarrow$(2) leg needs differentiability of Bernstein functions, and there is no route to
that which does not reintroduce $\CM$.
\textbf{The derivative is unavoidable here}, which is the other reason to keep it separate:
$B(s) = sF'(s)$ has no reading in $\LE$ until the representation is in hand, so unlike
\ref{lem:selfdecomposable-increment} this statement cannot be given an $\LE$ form at all. That
is a fact about the mathematics, not a gap in the transcription.)}
\end{lemma}

\begin{proof}
\emph{(1) $\Rightarrow$ (2).} For $h > 0$ the difference quotient
$s \mapsto \bigl(F(e^h s) - F(s)\bigr)/h$ lies in $\BF$, being a positive multiple of an
element of $\BFz$. As $h \downarrow 0$ it converges pointwise to $sF'(s)$, since $F$ is smooth
on $(0,\infty)$; the limit is therefore Bernstein by \ref{prop:bernstein-toolbox}(4).

\emph{(2) $\Rightarrow$ (1).} $F(bs) - F(as) = \int_{\log a}^{\log b} B(e^u s)\,du$. Each
integrand is Bernstein in $s$, so the finite mixture is Bernstein by
\ref{prop:bernstein-toolbox}(4); vanishing at $\zp$ follows by monotone convergence.

\emph{(2) $\Leftrightarrow$ (3).} Write $F(s) = b_0 s + \int (1 - e^{-st})\,\nu(dt)$, with no
killing term because $F(\zp) = 0$ (\ref{prop:bernstein-toolbox}(3)), so that
\[
  B(s) = s F'(s) = b_0 s + \int_0^\infty s\,t\, e^{-st}\, \nu(dt).
\]
Note $B(\zp) = 0$, since $0 \le sF'(s) \le F(s)$ by concavity.

Suppose $\nu(dt) = k(t)\,t^{-1}dt$ with $k$ nonincreasing. Write $k(t) = \rho\bigl((t,\infty)\bigr)$
for the positive measure $\rho = -dk$; right-continuity of $k$ may be assumed, and
$k(\infty) = 0$ is forced by $\int^\infty k(t)t^{-1}dt < \infty$. Tonelli then gives
\[
  \int_0^\infty s\, e^{-st}\, k(t)\, dt
  = \int_0^\infty \rho(du) \int_0^u s\,e^{-st}\,dt
  = \int_0^\infty \bigl(1 - e^{-su}\bigr)\,\rho(du),
\]
so $B(s) = b_0 s + \int (1 - e^{-su})\rho(du) \in \BFz$.

Conversely, if $B \in \BF$ then $B \in \BFz$ by the remark above, so
$B(s) = \beta s + \int (1 - e^{-su})\rho(du)$. Running the same computation backwards and
invoking uniqueness of the Lévy--Khintchine triple (\ref{prop:bernstein-toolbox}(3))
identifies $\beta = b_0$ and $\nu(dt) = k(t)t^{-1}dt$ with $k(t) = \rho((t,\infty))$
nonincreasing.
\end{proof}
